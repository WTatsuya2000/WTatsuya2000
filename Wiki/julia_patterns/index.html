    Jackknife 法

    よく使う3関数セット（JackKnife.jl として別ファイルに切り出して使う）。

    # NJK 番目のブロックのインデックス範囲
function BF(Ns::Int, Nbin::Int, NJK::Int)
    L     = div(Ns, Nbin)
    start = (NJK-1) * L + 1
    stop  = NJK * L
    return start:stop
end

# NJK 番目のブロックを除いた補集合
function BF_complement(Ns::Int, Nbin::Int, NJK::Int)
    L  = div(Ns, Nbin)
    r1 = 1:((NJK-1)*L)
    r2 = (NJK*L+1):Ns
    return [r for r in (r1, r2) if !isempty(r)]
end

# Bool マスク（そのブロックを false にする）
function BF_mask(Ns::Int, Nbin::Int, NJK::Int)
    mask = trues(Ns)
    mask[BF(Ns, Nbin, NJK)] .= false
    return mask
end

# JK 平均・誤差の計算（Nbin = 20 が典型）
function JackKnife(p0)
    Nbin  = length(p0)
    ave   = mean(p0, dims=1)
    var   = sum(p0.^2, dims=1) / Nbin .- ave.^2
    err   = sqrt.((Nbin-1) * var)
    return ave, err
end

    誤差伝播（加重平均）

    function error_propagation(values, errors)
    w   = @. 1 / errors^2
    W   = sum(w)
    w ./= W
    avg = sum(w .* values)
    return avg, sqrt(1/W)
end

# 偶・奇サイトをあらかじめグループに分けておく
G1, G2 = make_checkerboard_groups(Lx, Ly)

function sweep!(config, params)
    @threads for n in eachindex(G1)
        config[G1[n]...] = update_spin(G1[n], config, params)
    end
    @threads for n in eachindex(G2)
        config[G2[n]...] = update_spin(G2[n], config, params)
    end
end
