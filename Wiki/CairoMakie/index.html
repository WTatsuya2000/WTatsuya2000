    CairoMakie

    論文用図作成に使う描画ライブラリ。PDF/PNG/SVG を高品質で出力できる。

    インストール・インポート

    using Pkg; Pkg.add("CairoMakie")
using CairoMakie

    他パッケージと名前衝突が起きる場合は明示的に呼ぶ：

    CairoMakie.scatter!(ax, x, y; ...)

    基本構造

    fig = Figure(size=(800, 500))
ax  = Axis(fig[1, 1];
    xlabel = "x軸ラベル",
    ylabel = "y軸ラベル",
    title  = "タイトル",
)
# プロット...
save("output.png", fig; px_per_unit=2)  # 高解像度保存

    複数パネル：

    fig = Figure(size=(950, 700))
ax1 = Axis(fig[1, 1]; ...)
ax2 = Axis(fig[1, 2]; ...)
ax3 = Axis(fig[2, 1]; ...)

    軸の設定

    ax = Axis(fig[1, 1];
    limits       = (xmin, xmax, ymin, ymax),        # 描画範囲
    xticks       = (値の配列, ラベルの配列),          # 目盛り
    yticks       = ([0.0, 0.5, 1.0], ["0","0.5","1"]),
    xticklabelsize = 20,
    yticklabelsize = 20,
    xlabelsize   = 30,
    backgroundcolor = :transparent,
)

    凡例

    # シンプル
axislegend(ax; position=:rt, labelsize=13)

# ラベル指定（scatter等の戻り値を使う）
p1 = lines!(ax, x, y; color=:blue)
p2 = lines!(ax, x, z; color=:red)
axislegend(ax, [p1, p2], ["label1", "label2"];
    position = :rt,
    labelsize = 25,
    nbanks   = 2,   # 列数
)

    色の指定

    color = :blue                  # シンボル
color = (:blue, 0.5)           # 透明度付き (RGBA)
color = RGBf(0.2, 0.4, 0.8)   # RGB直接指定

# カラーパレット例（物理プロット用）
pal = cgrad(:tab10).colors     # 10色セット
color = pal[1]

    LaTeXラベル

    using CairoMakie  # L"..." マクロが使える

ax = Axis(fig[1,1];
    xlabel = L"T",
    ylabel = L"R_{21}(T, L)",
    title  = L"\chi^2 / \mathrm{d.o.f.}",
)

text!(ax, 0, 1; text = L"10^5 \Delta T = %$(val)")  # 変数埋め込み

    インセット（小図を図の中に埋め込む）

    ax_inset = Axis(fig[1, 1];
    width           = Relative(0.35),
    height          = Relative(0.45),
    halign          = 0.02,   # 左端
    valign          = 0.98,   # 上端
    backgroundcolor = :white,
    title           = "拡大",
    titlesize       = 11,
    xticklabelsize  = 9,
    yticklabelsize  = 9,
)
translate!(ax_inset.scene, 0, 0, 10)  # 前面に表示

    典型的なテンプレート（エラーバー付き比較プロット）

    using CairoMakie, Statistics

function jk_mean_err(v)
    N = length(v); N < 2 && return mean(v), NaN
    m = mean(v)
    θ = [(sum(v) - v[i])/(N-1) for i in 1:N]
    σ = sqrt((N-1)/N * sum((θ .- mean(θ)).^2))
    m, σ
end

fig = Figure(size=(800, 500))
ax  = Axis(fig[1, 1];
    xlabel = L"L",
    ylabel = L"R_{21}",
    title  = "Lee-Yang zero ratio",
)
hlines!(ax, [1.0]; color=:gray60, linewidth=1, linestyle=:dash)

colors  = [:dodgerblue, :tomato, :green]
markers = [:circle, :rect, :diamond]
offsets = [-0.15, 0.0, +0.15]

for (i, (Ls, Rs, errs, label)) in enumerate(datasets)
    xs = Ls .+ offsets[i]
    errorbars!(ax, xs, Rs, errs;
        color=colors[i], whiskerwidth=10, linewidth=2)
    scatter!(ax, xs, Rs;
        color=colors[i], marker=markers[i],
        markersize=14, label=label)
end

axislegend(ax; position=:rt, labelsize=13)
save("output.png", fig; px_per_unit=2)
