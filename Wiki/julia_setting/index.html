    Julia_Setting.jl

    ~/Julia_Setting.jl に置いてある共通設定ファイル。 各 notebook / スクリプトの冒頭で include して使う。

    include("/Users/wadatatsuya/Julia_Setting.jl")

    2. カスタムカラーパレット pal

    Set1_9 パレットをベースに、HSV空間で彩度・明度を微調整したカスタムパレット。

    set1_palette = palette(:Set1_9)

function adjust_palette_colors(palette)
    adjusted_colors = []
    for color in palette
        hsv_color    = HSV(color)
        adjusted_hsv = HSV(hsv_color.h,
                           1.0 - (1.0 - hsv_color.s) * 0.2,
                           hsv_color.v)          # 彩度を少し上げる
        push!(adjusted_colors, RGB(adjusted_hsv))
    end
    return adjusted_colors
end

pal = adjust_palette_colors(set1_palette)

# 個別調整
pal[6] = HSV(60,  1.0,  0.95)   # 黄色
pal[7] = HSV(200, 0.7,  1.0)    # 水色
pal[9] = HSV(50,  0.46, 0.87)   # ベージュ
push!(pal, HSV(0, 1.0, 0))      # pal[10] = 黒

    使い方

    plot!(x, y; color=pal[1])   # 赤系
plot!(x, y; color=pal[2])   # 青系
scatter!(x, y; c=pal[3])    # 緑系
# pal[10] = 黒

    3. 非線形方程式ソルバー nls

    NLsolve.jl のラッパー。スカラー・ベクトル両対応。

    using NLsolve

# --- シグネチャ ---
# nls(func, params...; ini=0.0, verbose=false)
#   func    : 残差関数 f(x, params...) = 0 を解く
#   params  : func に追加で渡す引数
#   ini     : 初期値（スカラーまたはベクトル）
#   verbose : 収束情報を表示するか
#
# 返り値: sol, r, converged

    使用例（スカラー）

    f(x, a) = x^2 - a          # x² = a を解く
sol, r, converged = nls(f, 2.0; ini=1.0)
println(sol)                # ≈ 1.4142...

    使用例（ベクトル）

    function F(v, p)
    x, y = v
    return [x^2 + y^2 - p, x - y]   # 連立方程式
end
sol, r, converged = nls(F, 1.0; ini=[0.5, 0.5])

    verbose モード

    sol, r, converged = nls(f, 2.0; ini=1.0, verbose=true)
# NLsolve summary
#   f_converged = true
#   x_converged = true
#   iterations  = 5
#   zero = [1.4142135...]

    4. 時刻表示 display_current_time

    using Dates

t = display_current_time()
println(t)   # 例: 2026-05-14T10:23:45.123

    計算の開始・終了時刻の記録に使う。

    t_start = display_current_time()
# ... 重い計算 ...
t_end = display_current_time()
println("経過時間: ", t_end - t_start)
